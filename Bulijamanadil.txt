import sys

def print_lyrics():
    lyrics = [
        "O thodach deen batas bhau tuti jai mana sapan",
        "Aar kawaj vhai jai mehndi rasam",
        "An kawaj lagi jai lagan...",
        "O thodach deen batas bhau tuti jai mana sapan",
        "Aar kawaj vhai jai mehndi rasam",

        "Bhulija mana dil tu tila re...",
        "Bhulija mana dil tu tila re...",
        "May tina ti mani ray ni nare...",
        "May tina ti mani ray ni nare..."
    ]
    
   
    delays = [
        3.5, 3.2, 4.0,
        3.5, 3.2, 4.5,
        3.8, 3.8,
        3.5, 4.0
    ]

    print("Bhulija Mana Dil (Ahirani Sad Song) - Animation\n")
    time.sleep(1.5)

    for i, line in enumerate(lyrics):
        for char in line:
            sys.stdout.write(char)
            sys.stdout.flush()
            
            time.sleep(0.08)
        
        print()
        if i < len(delays):
            time.sleep(delays[i])
