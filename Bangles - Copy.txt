    <script>
        const lyrics = [
            "Laal laal color chi baangdi",
            "Baangdicha color laal",
            "Chaal mazyasobat chaal",
            "Naach mazyasobat chaal",
            "Hey!",
            "Pandharya color cha mogra",
            "Kesala lavun hi chaal",
            "Chaal mazyasobat chaal",
            "Naach mazyasobat chaal",
            "",
            "",
            "",
            "",
            "",
            ""
        ];

        const container = document.getElementById('lyrics-container');
        let index = 0;

        function showNextLine() {
            const currentLine = container.querySelector('.lyric-line:not(.dying)');
            if (currentLine) {
                currentLine.classList.add('dying');
                const oldLetters = currentLine.querySelectorAll('.letter');
                oldLetters.forEach((l, i) => {
                    setTimeout(() => l.classList.add('exit'), i * 15);
                });
                setTimeout(() => currentLine.remove(), 800);
            }

            const lineDiv = document.createElement('div');
            lineDiv.className = 'lyric-line';
            
            const words = lyrics[index].split(' ');
            words.forEach(word => {
                const wordSpan = document.createElement('span');
                wordSpan.className = 'word';
                
                const lowerWord = word.toLowerCase();
                
                // Red Highlights
                if (lowerWord.includes('baangdi') || lowerWord === 'laal') {
                    wordSpan.classList.add('highlight-red');
                }
                // White/Mogra Highlights
                if (lowerWord.includes('mogra') || lowerWord.includes('pandharya')) {
                    wordSpan.classList.add('highlight-white');
                }
                // Hey! Special style
                if (lowerWord === 'hey!') {
                    lineDiv.classList.add('hey-style');
                }

                [...word].forEach(char => {
                    const span = document.createElement('span');
                    span.textContent = char;
                    span.className = 'letter';
                    wordSpan.appendChild(span);
                });
                
                lineDiv.appendChild(wordSpan);
            });

            container.appendChild(lineDiv);

            requestAnimationFrame(() => {
                const newLetters = lineDiv.querySelectorAll('.letter');
                newLetters.forEach((l, i) => {
                    setTimeout(() => l.classList.add('reveal'), i * 30);
                });
            });

            index = (index + 1) % lyrics.length;
        }

        showNextLine();
        setInterval(showNextLine, 3200);
    </script>
