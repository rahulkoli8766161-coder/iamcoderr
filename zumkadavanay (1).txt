import sys
import time

def print_lyrics():
    lyrics = [
        "Aathi dakhu mi Tathi dakhu",
        "Mana Zhukma padanay",
        "Zhumka Zhumka",
        "ar Zhumka davad nay ",
        "baya Zhumka davad nay",
        "Zhumka davad nay",
        "baya Zhumka davad nay",
        "(Zhumka davad nar)",
        "(ina Zhumka davad nar)",
        "(Zhumka davad nar)",
        "(ina Zhumka davad nar)"
    ]
    
    delays = [
        3.5, 3.8, 
        3.5, 2.8, 
        4.5, 3.8, 
        3.5, 2.2, 
        4.2, 3.8, 
        2.2, 3.5,
        2.4, 2.6,
        2,7, 2,5  
    ]

    print("Zhumka Davad Nay - Madhuri saidane \n")
    time.sleep(1.2)

    for i, line in enumerate(lyrics):
        for char in line:
            sys.stdout.write(char)
            sys.stdout.flush()
            
            time.sleep(0.06)
        
        print()
        if i < len(delays):
            time.sleep(delays[i])

if __name__ == "__main__":
    try:
        print_lyrics()
    except KeyboardInterrupt:
        print("\n[Stopped]")
